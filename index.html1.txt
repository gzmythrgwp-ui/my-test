<script>
// ==== БАЗА ВОПРОСОВ ====
const allQuestions = [
  { q: "Столица Франции?", a: ["Лондон", "Париж", "Берлин", "Мадрид"], correct: 1 },
  { q: "Сколько будет 7 × 8?", a: ["54", "56", "64", "48"], correct: 1 },
  { q: "Какая планета самая большая?", a: ["Земля", "Марс", "Юпитер", "Сатурн"], correct: 2 },
  { q: "Кто написал 'Войну и мир'?", a: ["Достоевский", "Толстой", "Пушкин", "Чехов"], correct: 1 },
  { q: "Химический символ золота?", a: ["Go", "Gd", "Au", "Ag"], correct: 2 },
  { q: "Сколько континентов на Земле?", a: ["5", "6", "7", "8"], correct: 2 },
  { q: "Самая длинная река в мире?", a: ["Амазонка", "Нил", "Янцзы", "Миссисипи"], correct: 0 },
  { q: "Кто изобрёл лампочку?", a: ["Тесла", "Эдисон", "Ньютон", "Эйнштейн"], correct: 1 },
  { q: "Сколько цветов в радуге?", a: ["5", "6", "7", "9"], correct: 2 },
  { q: "Язык программирования для веба?", a: ["Python", "JavaScript", "C++", "Java"], correct: 1 },
  { q: "Самое большое животное?", a: ["Слон", "Синий кит", "Жираф", "Бегемот"], correct: 1 },
  { q: "Год первого полёта человека в космос?", a: ["1957", "1961", "1969", "1975"], correct: 1 },
  { q: "Сколько сторон у шестиугольника?", a: ["5", "6", "7", "8"], correct: 1 },
  { q: "Кто написал 'Гарри Поттера'?", a: ["Толкин", "Роулинг", "Мартин", "Льюис"], correct: 1 },
  { q: "Главный герой 'Преступления и наказания'?", a: ["Пьер", "Раскольников", "Онегин", "Базаров"], correct: 1 },
];

// ==== ОСКОРБЛЕНИЯ ====
const insults = [
  "Ты вообще меня не знаешь…",
  "Я плохой. Я это признаю.",
  "Ты даже не пытаешься понять меня.",
  "Может, я и правда никуда не гожусь…",
  "Ты меня совсем не знаешь.",
  "Я разочаровываю всех. И тебя тоже.",
  "Ну кто я такой, чтобы ты меня слушал?",
  "Я не стою твоего внимания.",
];

// ==== СОСТОЯНИЕ ====
let usedQuestions = [];
let currentQuestion = null;
let wrongCount = 0;
const MAX_WRONG = 3;
const TOTAL_QUESTIONS = 10;
let askedCount = 0;

const qEl = document.getElementById('question');
const aEl = document.getElementById('answers');
const scoreEl = document.getElementById('score');
const insultEl = document.getElementById('insult');
const progressEl = document.getElementById('progress');

// Перемешать массив
function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// Получить новый вопрос (не из уже использованных в текущей сессии)
function getNextQuestion() {
  let pool = allQuestions.filter(q => !usedQuestions.includes(q));
  if (pool.length === 0) {
    // Если вопросы закончились — сбрасываем
    usedQuestions = [];
    pool = [...allQuestions];
  }
  const picked = pool[Math.floor(Math.random() * pool.length)];
  usedQuestions.push(picked);
  return picked;
}

function showInsult() {
  insultEl.textContent = insults[Math.floor(Math.random() * insults.length)];
}

function updateScore() {
  scoreEl.textContent = `Ошибок: ${wrongCount} / ${MAX_WRONG}  |  Вопрос: ${askedCount}`;
  progressEl.style.width = (askedCount / TOTAL_QUESTIONS * 100) + '%';
}

function loadQuestion() {
  if (askedCount >= TOTAL_QUESTIONS && wrongCount < MAX_WRONG) {
    finish(true);
    return;
  }
  if (wrongCount >= MAX_WRONG) {
    // Превышен лимит — сбрасываем вопросы и продолжаем
    wrongCount = 0;
    askedCount = 0;
    usedQuestions = [];
    insultEl.textContent = "";
    // Показать сообщение
    qEl.textContent = "Слишком много ошибок. Начнём заново…";
    aEl.innerHTML = "";
    setTimeout(() => {
      askedCount = 0;
      loadQuestion();
    }, 1500);
    return;
  }

  currentQuestion = getNextQuestion();
  askedCount++;
  updateScore();

  qEl.textContent = currentQuestion.q;
  aEl.innerHTML = "";
  insultEl.textContent = "";

  // Перемешиваем ответы, но запоминаем правильный
  const indexed = currentQuestion.a.map((text, i) => ({ text, isCorrect: i === currentQuestion.correct }));
  const shuffled = shuffle(indexed);

  shuffled.forEach((ans, i) => {
    const btn = document.createElement('button');
    btn.textContent = ans.text;
    btn.onclick = () => handleAnswer(btn, ans.isCorrect);
    aEl.appendChild(btn);
  });
}

function handleAnswer(btn, isCorrect) {
  // Блокируем все кнопки
  [...aEl.children].forEach(b => {
    b.disabled = true;
    if (b.textContent === currentQuestion.a[currentQuestion.correct]) {
      b.classList.add('correct');
    }
  });

  if (!isCorrect) {
    btn.classList.add('wrong');
    wrongCount++;
    updateScore();
    showInsult();

    if (wrongCount >= MAX_WRONG) {
      setTimeout(() => {
        loadQuestion(); // сброс произойдёт внутри
      }, 1800);
      return;
    }
  } else {
    insultEl.textContent = "";
  }

  setTimeout(loadQuestion, 1200);
}

function finish(success) {
  qEl.textContent = success ? "✅ Ты прошёл тест!" : "❌ Попробуй ещё раз";
  aEl.innerHTML = "";
  insultEl.textContent = success ? "Ладно… может, ты меня и знаешь." : "";
}

// Запуск
loadQuestion();
